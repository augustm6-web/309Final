# store models in list 
models <- list(
  "Logistic (all)" = fit_glm_all,
  "Logistic (subset)" = fit_glm_small,
  "Decision Tree" = fit_tree,
  "Random Forest" = fit_rf,
  "k-NN" = fit_knn
)

# cross val results
cv_results <- lapply(names(models), function(name) {
  fit <- models[[name]]
  best <- merge(fit$bestTune, fit$results)
  data.frame(
    Model = name,
    ROC = best$ROC[1],
    Sens = best$Sens[1],
    Spec = best$Spec[1]
  )
}) %>%
  bind_rows()


# test results 
test_results <- lapply(names(models), function(name) {
  fit <- models[[name]]
  preds <- predict(fit, newdata = bird_test)
  probs <- predict(fit, newdata = bird_test, type = "prob")[,"Yes"]

  cm <- confusionMatrix(
    preds,
    bird_test$feeding,
    positive = "Yes"
  )

  auc_val <- auc(
    roc(bird_test$feeding, probs,
        levels = c("No", "Yes"))
  )

  data.frame(
    Model = name,
    Accuracy = cm$overall["Accuracy"],
    AUC = as.numeric(auc_val),
    Sensitivity = cm$byClass["Sensitivity"],
    Specificity = cm$byClass["Specificity"]
  )

}) %>%
  bind_rows()
